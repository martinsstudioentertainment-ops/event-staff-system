Appendix A — SQL Recommendations (do not execute without approval)

-- DIAGNOSTIC (read-only)
SELECT table_name, engine, table_rows,
       ROUND(data_length/1024,1) AS data_kb,
       ROUND(index_length/1024,1) AS index_kb
FROM information_schema.tables
WHERE table_schema = DATABASE()
ORDER BY data_length DESC;

EXPLAIN SELECT COUNT(*) FROM staff_registrations
WHERE event_id=1 AND status='pending';

-- ORPHAN RE-CHECK (expect 0)
SELECT COUNT(*) FROM staff_registrations
WHERE staff_id IS NULL OR staff_id=0;

-- OPTIMIZATION (change window)
ALTER TABLE staff_registrations
  ADD COLUMN email_normalized VARCHAR(150)
    GENERATED ALWAYS AS (LOWER(TRIM(email))) STORED,
  ADD INDEX idx_sr_email_normalized (email_normalized);

-- CLEANUP (P18 — review first)
DELETE FROM ip_geo_cache
